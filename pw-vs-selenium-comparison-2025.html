🎭 Playwright — JavaScript / TypeScript
✅ Clean · Auto-wait · No boilerplate · Built-in assertions
// ✅ Install: npm init playwright@latest  (one command!)
// ✅ No WebDriverManager, no chromedriver downloads needed!
import { test, expect } from '@playwright/test';

test('Login with valid credentials', async ({ page }) => {

  // Navigate — auto-waits for page load
  await page.goto('https://app.example.com/login');

  // Fill fields — auto-waits for visibility!
  await page.fill('#username', 'admin@test.com');
  await page.fill('#password', 'Admin@123');

  // Click button — auto-waits until clickable!
  await page.click('button[type="submit"]');

  // Assertion — built-in, no extra imports!
  await expect(page).toHaveURL(/dashboard/);
  await expect(page.getByText('Welcome Back'))
                   .toBeVisible();
});

// Run: npx playwright test
// HTML Report auto-generated → open in browser ✨
// Screenshots + Videos auto-recorded on failure 🎬

🌐 Selenium — Java + TestNG
⚠️ More setup · Manual waits required · Separate libraries needed
// ⚠️ Need Maven pom.xml + Selenium + TestNG dependencies
// ⚠️ Need ExtentReports/Allure for HTML reports (extra work!)
import org.openqa.selenium.*;
import org.openqa.selenium.chrome.*;
import org.openqa.selenium.support.ui.*;
import org.testng.Assert;
import org.testng.annotations.*;

public class LoginTest {
  WebDriver driver;
  WebDriverWait wait;

  @BeforeMethod
  public void setup() {
    driver = new ChromeDriver(new ChromeOptions());
    driver.manage().window().maximize();
    wait = new WebDriverWait(driver,
                Duration.ofSeconds(15));
  }

  @Test
  public void loginTest() {
    driver.get("https://app.example.com/login");

    // Find elements manually every time
    driver.findElement(By.id("username"))
          .sendKeys("admin@test.com");
    driver.findElement(By.id("password"))
          .sendKeys("Admin@123");

    // Must explicitly wait — or it may fail!
    wait.until(ExpectedConditions
      .elementToBeClickable(
        By.cssSelector("button[type='submit']")))
      .click();

    wait.until(ExpectedConditions
      .urlContains("dashboard"));
    Assert.assertTrue(
      driver.getCurrentUrl().contains("dashboard"));
  }

  @AfterMethod
  public void teardown() { driver.quit(); }
}
// HTML Report → set up Allure/Extent separately

Page Object Model — Industry-Wide Standard
// POM pattern — every company uses this!
public class LoginPage {
  @FindBy(id="username") WebElement usernameField;
  @FindBy(id="password") WebElement passwordField;
  @FindBy(id="loginBtn")  WebElement loginButton;

  public void login(String u, String p) {
    usernameField.sendKeys(u);
    passwordField.sendKeys(p);
    loginButton.click();    // Reusable ♻️
  }
}
